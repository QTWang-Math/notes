\usepackage[
  backend=biber,
  style=alphabetic,
  sorting=anyt,
  maxalphanames=4,
  maxnames=99,
  giveninits=true,
  doi=true,
  eprint=true,
  url=true,
  urldate=iso,
  seconds=true,
  isbn=false,
  backref=true
]{biblatex}

\addbibresource{references.bib}

\DeclareFieldFormat[article]{eid}{Art\adddot\space#1}
\DeclareFieldFormat{pagetotal}{#1\addspace pp\adddot}

% Journal articles name the journal directly; chapters retain "In:".
\renewbibmacro*{in:}{%
  \ifentrytype{article}
    {}
    {\printtext{\bibstring{in}\intitlepunct}}%
}

% Give one access link: DOI, otherwise arXiv/eprint, otherwise the URL.
\renewbibmacro*{doi+eprint+url}{
  \iffieldundef{doi}
    {\iffieldundef{eprint}
       {\usebibmacro{url+urldate}}
       {\usebibmacro{eprint}}}
    {\printfield{doi}}
}

\DefineBibliographyStrings{english}{
  backrefpage  = {$\uparrow$},
  backrefpages = {$\uparrow$}
}

% Long titles and identifiers should wrap instead of entering the margin.
\AtBeginBibliography{\small\sloppy}
